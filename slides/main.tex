\frametitle{Motivation: Why Search is Critical in Modern Systems}
\begin{columns}[T]
\begin{column}{0.52\textwidth}
\begin{block}{Modern Data Scale \& User Expectations}
\small
\begin{itemize}
    \item \textbf{E-Commerce Catalogs:} Millions of products; users expect sub-20ms responses, typo tolerance, faceted filters, and semantic ranking.
    \item \textbf{Observability \& Logs:} Petabytes of semi-structured events requiring real-time ad-hoc text search and aggregations.
    \item \textbf{Social Media \& Messaging:} Billions of messages indexed simultaneously by user/channel key and full-text content.
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.45\textwidth}
\begin{alertblock}{Why Traditional RDBMS Struggle}
\small
\begin{itemize}
    \item \texttt{WHERE description LIKE '\%laptop\%'} forces a \textbf{full table scan} $O(N)$.
    \item B$^+$-trees only index exact left-prefixes (\texttt{'laptop\%'}), not arbitrary interior tokens.
    \item No built-in \textbf{relevance scoring} (all matching rows returned with equal weight).
    \item Vertical scaling bottlenecks under heavy read/write concurrency.
\end{itemize}
\end{alertblock}
\vspace{0.05cm}
\small
\textbf{Core Question:} How do \textbf{Elasticsearch 8.15} (search engine) and \textbf{Cassandra 5.0} (wide-column store) solve retrieval at scale?
\end{column}
\end{columns}
\end{frame}

% SLIDE 5: Search Fundamentals
%---------------------------------------------------------
\begin{frame}
\frametitle{Search Fundamentals: Inverted Index \& Ranking Formulas}
\begin{columns}[T]
\begin{column}{0.46\textwidth}
\textbf{Inverted Index Structure} \cite{inverted}
\scriptsize
\begin{itemize}
    \item Maps each unique \textbf{term} (in a sorted Term Dictionary / FST) to a \textbf{posting list} of document IDs, term frequencies, and positions.
\end{itemize}
\vspace{0.05cm}
\centering
\tiny
\begin{tabular}{lll}
\toprule
\textbf{Term} & \textbf{DF} & \textbf{Posting List (Doc : TF)} \\
\midrule
\texttt{camera}     & 8,461 & $[D_{12}: 2, \; D_{45}: 1, \dots]$ \\
\texttt{grinder}    & 3,716 & $[D_{5212}: 2, \; D_{6010}: 1, \dots]$ \\
\texttt{waterproof} & 6,140 & $[D_{104}: 1, \; D_{22775}: 1, \dots]$ \\
\bottomrule
\end{tabular}
\vspace{0.1cm}
\raggedright
\scriptsize
\begin{itemize}
    \item \textbf{FST (Finite State Transducer):} Compresses term dictionary in RAM for fast prefix and fuzzy automaton traversal.
    \item \textbf{Skip Lists \& Roaring Bitmaps:} Accelerate boolean conjunctions ($\cap$) and cached filter bitsets.
\end{itemize}
\end{column}
\begin{column}{0.51\textwidth}
\textbf{From TF-IDF to Okapi BM25} \cite{bm25}
\scriptsize
\begin{itemize}
    \item \textbf{TF-IDF:} $\text{TF}(t,d) \times \text{IDF}(t)$ grows unboundedly with term frequency.
    \item \textbf{BM25} adds \textbf{term saturation} ($k_1$) and \textbf{document length normalization} ($b$):
\end{itemize}
\vspace{-0.2cm}
{\tiny
\[
\text{BM25}(D, Q) = \sum_{t \in Q} \text{IDF}(t) \cdot \frac{f(t, D) \cdot (k_1 + 1)}{f(t, D) + k_1 \cdot \left(1 - b + b \cdot \frac{|D|}{\text{avgdl}}\right)}
\]
\[
\text{IDF}(t) = \ln\!\left(1 + \frac{N - n(t) + 0.5}{n(t) + 0.5}\right)
\]
}
\scriptsize
\begin{itemize}
    \item $k_1 = 1.2$: controls TF saturation asymptote (note: Lucene 8+ folds $(k_1+1)=2.2$ into \texttt{boost}).
    \item $b = 0.75$: penalizes verbose documents ($|D| > \text{avgdl}$).
\end{itemize}
\end{column}
\end{columns}
\end{frame}

\frametitle{Elasticsearch Overview \& Distributed Architecture}
\begin{columns}[T]
\begin{column}{0.52\textwidth}
\textbf{Core Architectural Concepts} \cite{elasticsearch,elastic_docs}
\scriptsize
\begin{itemize}
    \item \textbf{Cluster \& Nodes:} A collection of nodes coordinating storage and distributed search execution.
    \item \textbf{Index:} Logical namespace mapping to one or more physical shards.
    \item \textbf{Shard (Lucene Instance):} Self-contained Apache Lucene index consisting of immutable \textbf{segments}.
    \begin{itemize}
        \scriptsize
        \item \textbf{Primary Shard:} Handles indexing writes first.
        \item \textbf{Replica Shard:} Redundant copy for high availability and read throughput scaling.
    \end{itemize}
    \item \textbf{Scatter--Gather Search:} Coordinating node broadcasts query to shards, merges top-$K$ scores.
\end{itemize}
\end{column}

\begin{columns}[T]
\begin{column}{0.49\textwidth}
\begin{block}{Four Co-Located Per-Segment Structures}
\scriptsize
\begin{enumerate}
    \item \textbf{Inverted Index (+ FST):} Token $\rightarrow$ sorted posting list of \texttt{(doc\_id, tf, positions)} for full-text search.
    \item \textbf{Doc Values (Columnar):} Column-stride arrays on disk (\texttt{.dvd/.dvm}) for fast sorting \& aggregations.
    \item \textbf{BKD Trees:} Block K-D trees for $O(\log N)$ numeric, date, and geo-spatial range filtering.
    \item \textbf{HNSW Graph (\texttt{.vec/.vex}):} Multi-layer proximity graph for dense vector $k$-NN similarity search.
\end{enumerate}
\end{block}
\end{column}
\begin{column}{0.47\textwidth}
\begin{block}{Near Real-Time (NRT) Refresh \& Durability}
\scriptsize
\begin{itemize}
    \item \textbf{In-Memory Buffer + Translog:} New documents are appended to the write-ahead \texttt{translog} for crash recovery and buffered in RAM.
    \item \textbf{Refresh (every 1s):} Buffer is written to a new immutable \textbf{Lucene segment} in OS page cache $\rightarrow$ document becomes searchable within 1 second.
    \item \textbf{Flush \& Tiered Merge:} Periodically commits segments to disk (\texttt{fsync}), clears translog, and merges smaller segments in the background while purging deleted docs.
\end{itemize}
\end{block}
\end{column}
\end{columns}
\end{frame}

  { "value": 3.7997048,

  { "value": 4.0027270,
    "description": "weight(description:grinder) [f=2, n=3716]",
    "details": [
      { "value": 2.2, "description": "boost (k1+1)" },
      { "value": 2.5992606, "description": "idf (n=3716, N=50000)" },
      { "value": 0.6999765, "description": "tf (f=2, dl=15, avgdl=24.23)" }
    ]} ]}
\end{lstlisting}
\end{column}
\begin{column}{0.47\textwidth}
\textbf{\tiny Exact Mathematical Verification ($N = 50,000$):}
\tiny
Doc $\text{\_id}=5212$ has $|D|=15$ vs. $\text{avgdl}=24.2266$, so length norm $K = 1.2\left(0.25 + 0.75\frac{15.0}{24.2266}\right) = \mathbf{0.85723}$.
\begin{itemize}
    \item \textbf{Terms \texttt{conic} \& \texttt{burr}} ($f=1,\; n=2022$):
    \vspace{-0.12cm}
    \[
    \text{IDF} = \ln\!\left(1 + \frac{50000 - 2022 + 0.5}{2022 + 0.5}\right) = 3.20771
    \]
    \vspace{-0.22cm}
    \[
    \text{TF}_{\text{norm}} = \frac{1.0}{1.0 + 0.85723} = 0.53843, \quad S = 2.2 \times \text{IDF} \times \text{TF}_{\text{norm}} = \mathbf{3.79970}
    \]
    \item \textbf{Term \texttt{grinder}} ($f=2,\; n=3716$):
    \vspace{-0.12cm}
    \[
    \text{IDF} = \ln\!\left(1 + \frac{50000 - 3716 + 0.5}{3716 + 0.5}\right) = 2.59926
    \]
    \vspace{-0.22cm}
    \[
    \text{TF}_{\text{norm}} = \frac{2.0}{2.0 + 0.85723} = 0.69998, \quad S = 2.2 \times \text{IDF} \times \text{TF}_{\text{norm}} = \mathbf{4.00273}
    \]
\end{itemize}
\vspace{-0.12cm}
\begin{block}{\tiny Total Score \& TF Saturation Insight}
\tiny
$\text{Total BM25} = 3.79970 + 3.79970 + 4.00273 = \mathbf{11.602137}$. Doubling \texttt{grinder} ($f: 1 \rightarrow 2$) raises $\text{TF}_{\text{norm}}$ by only $\mathbf{+30\%}$ ($0.538 \rightarrow 0.700$)!
\end{block}
\end{column}
\end{columns}

\begin{column}{0.48\textwidth}
\begin{block}{\scriptsize Standard Two-Phase \texttt{query\_then\_fetch}}
\scriptsize
\begin{enumerate}
    \item \textbf{Query Phase:} Coordinating node fans out query to all $P$ shards. Each shard executes locally and returns \emph{only} top-$K$ lightweight tuples \texttt{(\_id, \_score)}.
    \item \textbf{Fetch Phase:} Coordinator merges $P \times K$ scores, picks global top-$K$, and requests full \texttt{\_source} JSON only from the winning shards.
\end{enumerate}
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{alertblock}{\scriptsize The Local IDF Skew Problem \& \texttt{dfs\_query\_then\_fetch}}
\scriptsize
\begin{itemize}
    \item By default, each shard computes $\text{IDF}(t)$ using \textbf{local shard statistics} ($N_i, n_i$) for speed.
    \item On small or custom-routed indices, term frequencies skew across shards $\rightarrow$ identical documents on different shards get different BM25 scores!
    \item \textbf{Solution:} \texttt{search\_type=dfs\_query\_then\_fetch} adds a pre-query round-trip to aggregate global $\sum N_i$ and $\sum n_i$ before scoring.
\end{itemize}
\end{alertblock}
\end{column}
\end{columns}
\end{frame}

\begin{column}{0.54\textwidth}
\textbf{Decentralized Peer-to-Peer Design} \cite{cassandra,consistent_hashing}
\scriptsize
\begin{itemize}
    \item \textbf{Masterless Ring Topology:} Every node plays an identical role; no single point of failure (SPOF).
    \item \textbf{Consistent Hashing:} Partition key is hashed via \textbf{Murmur3Partitioner} ($\text{range } [-2^{63}, 2^{63}-1]$) to determine token ownership on the ring.

    \item \textbf{Tunable Consistency:} Configure \texttt{ONE}, \texttt{LOCAL\_QUORUM}, or \texttt{ALL} per query ($R + W > \text{RF}$ guarantees strong consistency).
\end{itemize}

\begin{columns}[T]
\begin{column}{0.49\textwidth}
\begin{alertblock}{\tiny 1. The Wide-Partition Hotspot Problem}
\tiny
\begin{itemize}

\begin{lstlisting}[language=SQL]

Distributes \texttt{'laptops'} across multiple vnodes while preserving sorted price scans inside each bucket!

\end{itemize}
\end{alertblock}

\begin{frame}
\frametitle{2026 Research Frontier: Dense Vector \& Hybrid Search}
\begin{columns}[T]
\begin{column}{0.49\textwidth}
\textbf{\scriptsize Approximate Nearest Neighbor (ANN) Graphs}
\tiny
\begin{itemize}
    \item Lexical BM25 fails on \textbf{vocabulary mismatch} (e.g., querying \texttt{"portable commuter gear"} vs. product text \texttt{"lightweight ultrabook"}).
    \item Dense embeddings map items to $\mathbf{u}, \mathbf{v} \in \mathbb{R}^d$ scored by \textbf{Cosine Similarity}:
\end{itemize}
\vspace{-0.15cm}
{\tiny
\[
\text{sim}_{\cos}(\mathbf{u}, \mathbf{v}) = \frac{\mathbf{u} \cdot \mathbf{v}}{\|\mathbf{u}\|_2 \, \|\mathbf{v}\|_2}, \qquad S_{\text{ES}}(\mathbf{u}, \mathbf{v}) = \frac{1 + \text{sim}_{\cos}(\mathbf{u}, \mathbf{v})}{2}
\]
}
\vspace{-0.12cm}
\tiny
\begin{itemize}
    \item \textbf{Elasticsearch 8.15 (Lucene HNSW, Malkov \& Yashunin 2020):} Multi-layer hierarchical skip-graph ($M=16$, $\text{efConstruction}=100$). Fast $O(\log N)$ greedy routing from coarse top layer to dense bottom layer $L_0$.
    \item \textbf{Cassandra 5.0 SAI (JVector / DiskANN, Subramanya et al. 2019):} Single-layer \textbf{Vamana graph} with Product Quantization (\texttt{CompressedVectors.db}), enabling disk-resident billion-scale ANN search without holding full vectors in JVM heap!
\end{itemize}
\end{column}
\begin{column}{0.48\textwidth}
\textbf{\scriptsize Hybrid Search Fusion: Linear Boost vs. RRF}
\tiny
\begin{itemize}
    \item Combining lexical BM25 ($S_{\text{BM25}}$) with semantic vector similarity ($S_{\text{kNN}}$) captures both \textbf{exact SKU/brand matches} and \textbf{conceptual intent} (Cormack et al., SIGIR 2009):
\end{itemize}
\vspace{-0.12cm}
{\tiny
\[
S_{\text{Hybrid}}(d) = w_{\text{lex}} \cdot S_{\text{BM25}}(d) + w_{\text{vec}} \cdot S_{\text{kNN}}(d)
\]
\[
\text{RRF}(d) = \sum_{m \in \{\text{BM25}, \text{kNN}\}} \frac{1}{k + \text{rank}_m(d)} \quad (k = 60)
\]
}

\begin{frame}
\frametitle{Dual-Store Consistency Engineering: Solving the NRT \& CDC Lag}
\begin{columns}[T]
\begin{column}{0.51\textwidth}
\begin{alertblock}{\tiny The Eventual Consistency Race Condition}
\tiny
\begin{itemize}
    \item Total sync delay $\Delta t = t_{\text{CDC}} + t_{\text{Kafka}} + t_{\text{ES\_refresh}} \approx 1.2\text{--}2.5\text{ seconds}$.
    \item \textbf{Scenario:} Last unit of \texttt{Sony AlphaShot} ($\text{stock}=1$) is purchased at $t=0$. At $t=0.5\text{s}$, Elasticsearch still shows $\text{stock}=1$! How do we prevent overselling?
\end{itemize}
\end{alertblock}
\begin{block}{\tiny Solution: Read-Repair \& Authoritative Verification}
\tiny
\begin{enumerate}
    \item \textbf{Search Stage (Eventual Consistency OK):} ES returns top-$K$ candidate IDs (\texttt{[26015, 13815, 16040]}).
    \item \textbf{Hydration Stage (Fast Point Lookup):} API Gateway fetches live price/stock from Cassandra at \texttt{LOCAL\_QUORUM} ($1.7\text{ ms}$).
    \item \textbf{Checkout Stage (Linearizable LWT):} Atomic compare-and-set in Cassandra:
    \texttt{UPDATE inventory SET stock = stock - 1 WHERE product\_id = 26015 IF stock > 0;}
\end{enumerate}
\end{block}

\scriptsize
\begin{itemize}
    \item Answers \textbf{known, high-velocity access patterns} in $1.70\text{ ms}$ ($O(1)$ routing) via consistent hashing and append-only LSM-Tree partitions.
    \item Cassandra 5.0 \textbf{SAI} adds low-overhead Trie, BKD-tree, and JVector ANN filtering, while guarding cluster SLAs against full-scan anti-patterns.
\end{itemize}
\end{block}

    \item \textbf{Production Standard:} Pair \textbf{Cassandra as the authoritative source of truth} (\texttt{LOCAL\_QUORUM} + LWT checkout) with \textbf{Elasticsearch as the discovery engine}, synced via Kafka CDC.
\end{itemize}
\end{alertblock}
