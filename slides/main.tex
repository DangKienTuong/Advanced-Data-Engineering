\begin{frame}
\frametitle{Motivation: Why Search is Critical in Modern Systems}
\begin{columns}[T]
\begin{column}{0.52\textwidth}
\begin{block}{Modern Data Scale \& User Expectations}
\small
\begin{itemize}
    \item \textbf{E-Commerce Catalogs:} Millions of products; users expect sub-100ms responses, typo tolerance, faceted filters, and personalized ranking.
    \item \textbf{Observability \& Logs:} Petabytes of semi-structured events requiring real-time ad-hoc text search and aggregations.
    \item \textbf{Social Media \& Messaging:} Billions of messages indexed simultaneously by user/channel key and full-text content.
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.45\textwidth}
\begin{alertblock}{Why Traditional RDBMS Struggle}
\small
\begin{itemize}
    \item \texttt{WHERE description LIKE '\%laptop\%'} forces a \textbf{full table scan} $O(N)$.
    \item B-trees only index exact prefixes (\texttt{'laptop\%'}), not arbitrary tokens.
    \item No built-in \textbf{relevance scoring} (all matching rows returned with equal weight).
    \item Vertical scaling bottlenecks under heavy read/write concurrency.
\end{itemize}
\end{alertblock}
\vspace{0.05cm}
\small
\textbf{Core Question:} How do \textbf{Elasticsearch} (search engine) and \textbf{Cassandra} (wide-column store) solve retrieval at scale?
\end{column}
\end{columns}
\end{frame}

%---------------------------------------------------------
% SLIDE 4: Search Fundamentals
%---------------------------------------------------------
\begin{frame}
\frametitle{Search Fundamentals: Inverted Index \& Ranking Formulas}
\begin{columns}[T]
\begin{column}{0.46\textwidth}
\textbf{Inverted Index Structure} \cite{inverted}
\scriptsize
\begin{itemize}
    \item Maps each unique \textbf{term} (in a sorted Term Dictionary / FST) to a \textbf{posting list} of document IDs, term frequencies, and positions.
\end{itemize}
\vspace{0.05cm}
\centering
\tiny
\begin{tabular}{lll}
\toprule
\textbf{Term} & \textbf{DF} & \textbf{Posting List (Doc : TF)} \\
\midrule
\texttt{camera}     & 2 & $[D_1: 2, \; D_3: 1]$ \\
\texttt{laptop}     & 2 & $[D_2: 1, \; D_4: 3]$ \\
\texttt{waterproof} & 1 & $[D_1: 1]$ \\
\bottomrule
\end{tabular}
\vspace{0.1cm}
\raggedright
\scriptsize
\begin{itemize}
    \item \textbf{FST (Finite State Transducer):} Compresses term dictionary in memory for fast prefix and fuzzy traversal.
    \item \textbf{Skip Lists / Roaring Bitmaps:} Accelerate boolean conjunctions ($\cap$) and disjunctions ($\cup$).
\end{itemize}
\end{column}
\begin{column}{0.51\textwidth}
\textbf{From TF-IDF to Okapi BM25} \cite{bm25}
\scriptsize
\begin{itemize}
    \item \textbf{TF-IDF:} $\text{TF}(t,d) \times \text{IDF}(t)$ grows unboundedly with term frequency.
    \item \textbf{BM25} adds \textbf{term saturation} ($k_1$) and \textbf{document length normalization} ($b$):
\end{itemize}
\vspace{-0.2cm}
{\tiny
\[
\text{BM25}(D, Q) = \sum_{t \in Q} \text{IDF}(t) \cdot \frac{f(t, D) \cdot (k_1 + 1)}{f(t, D) + k_1 \cdot \left(1 - b + b \cdot \frac{|D|}{\text{avgdl}}\right)}
\]
\[
\text{IDF}(t) = \ln\!\left(1 + \frac{N - n(t) + 0.5}{n(t) + 0.5}\right)
\]
}
\scriptsize
\begin{itemize}
    \item $k_1 \approx 1.2$: controls TF saturation asymptote.
    \item $b \approx 0.75$: penalizes long documents ($|D| > \text{avgdl}$).
\end{itemize}
\end{column}
\end{columns}
\end{frame}

\frametitle{Elasticsearch Overview \& Distributed Architecture}
\begin{columns}[T]
\begin{column}{0.52\textwidth}
\textbf{Core Architectural Concepts} \cite{elasticsearch,elastic_docs}
\scriptsize
\begin{itemize}
    \item \textbf{Cluster \& Nodes:} A collection of nodes coordinating storage and distributed search execution.
    \item \textbf{Index:} Logical namespace mapping to one or more physical shards.
    \item \textbf{Shard (Lucene Instance):} Self-contained Apache Lucene index consisting of immutable \textbf{segments}.
    \begin{itemize}
        \scriptsize
        \item \textbf{Primary Shard:} Handles indexing writes first.
        \item \textbf{Replica Shard:} Redundant copy for high availability and read throughput scaling.
    \end{itemize}
    \item \textbf{Scatter--Gather Search:} Coordinating node broadcasts query to shards, merges top-$K$ scores.
\end{itemize}
\end{column}

\vspace{0.15cm}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{block}{From JSON to Searchable Structures}
\scriptsize
\begin{enumerate}
    \item \textbf{Document Ingestion:} JSON fields parsed according to index mapping (\texttt{text}, \texttt{keyword}, \texttt{float}).
    \item \textbf{Analysis:} \texttt{text} fields pass through Character Filters $\rightarrow$ Tokenizer $\rightarrow$ Token Filters.
    \item \textbf{Multi-Representation:}
    \begin{itemize}
        \scriptsize
        \item \textbf{Inverted Index:} Token $\rightarrow$ Posting list.
        \item \textbf{Doc Values:} Column-stride storage for sorting and aggregations.
        \item \textbf{BKD Trees:} Multi-dimensional numeric/range indexing.
    \end{itemize}
\end{enumerate}
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{block}{Near Real-Time (NRT) Refresh \& Durability}
\scriptsize
\begin{itemize}
    \item \textbf{In-Memory Buffer + Translog:} New documents are appended to the write-ahead \texttt{translog} for crash recovery and buffered in RAM.
    \item \textbf{Refresh (every 1s):} Buffer is written to a new immutable \textbf{Lucene segment} in OS page cache $\rightarrow$ document becomes searchable.
    \item \textbf{Flush \& Segment Merge:} Periodically commits segments to disk (\texttt{fsync}), clears translog, and merges smaller segments in the background.
\end{itemize}
\end{block}
\end{column}
\end{columns}
\end{frame}

%---------------------------------------------------------

\begin{frame}
\frametitle{Apache Cassandra Overview: Masterless Ring Architecture}
\begin{columns}[T]
\begin{column}{0.54\textwidth}
\textbf{Decentralized Peer-to-Peer Design} \cite{cassandra,consistent_hashing}
\scriptsize
\begin{itemize}
    \item \textbf{Masterless Ring Topology:} Every node plays an identical role; no single point of failure (SPOF).
    \item \textbf{Consistent Hashing:} Partition key is hashed via \textbf{Murmur3Partitioner} ($\text{range } [-2^{63}, 2^{63}-1]$) to determine token ownership on the ring.
    \item \textbf{Virtual Nodes (vnodes):} Each physical node owns multiple token ranges (e.g., 16--256 vnodes) for balanced data distribution and seamless cluster resizing.
    \item \textbf{Coordinator Node:} Any node can receive a client request and route it directly in $O(1)$ hops to the replica nodes holding that token.
    \item \textbf{Tunable Consistency:} Configure \texttt{ONE}, \texttt{QUORUM}, or \texttt{ALL} per query ($R + W > \text{RF}$ guarantees strong consistency).
\end{itemize}
\end{column}
\begin{column}{0.43\textwidth}
\centering

\vspace{0.15cm}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{block}{Why Writes Are Ultra-Fast ($O(1)$ Disk I/O)}
\scriptsize
\begin{itemize}
    \item \textbf{No Read-Before-Write:} Unlike B-Trees, Cassandra never seeks or overwrites existing pages in place \cite{lsm}.
    \item \textbf{Dual Write:} Sequential append to the \textbf{Commit Log} (durability) + insertion into the in-memory \textbf{Memtable} sorted by clustering key.
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{block}{SSTables, Tombstones \& Compaction}
\scriptsize
\begin{itemize}
    \item \textbf{SSTable (Sorted String Table):} Flushed sequentially to disk; completely immutable.
    \item \textbf{Deletes = Tombstones:} Marker written with timestamp; reclaimed after \texttt{gc\_grace\_seconds}.
    \item \textbf{Compaction:} Periodically merges multiple SSTables into one, keeping only the latest cell timestamp.
\end{itemize}
\end{block}
\end{column}
\end{columns}
\end{frame}

\begin{columns}[T]
\begin{column}{0.52\textwidth}
\begin{block}{How Cassandra Avoids Reading Every SSTable}
\scriptsize
\begin{enumerate}
    \item \textbf{Bloom Filter (RAM):} Probabilistic bit array per SSTable. Guarantees \emph{no false negatives} --- immediately skips SSTables that definitely do not contain the partition key.
    \item \textbf{Key Cache \& Partition Summary:} Jumps directly to the byte offset in the Partition Index without extra disk seeks.
    \item \textbf{Clustering Slice Optimization:} Min/max clustering column metadata skips SSTables outside the requested range (e.g., \texttt{price BETWEEN 500 AND 800}).
\end{enumerate}
\end{block}

% SLIDE 14: Cassandra Query-First Design
%---------------------------------------------------------
\begin{frame}[fragile]
\frametitle{Cassandra Query-First Data Modeling}
\begin{columns}[T]
\begin{column}{0.52\textwidth}
\textbf{\scriptsize Schema Definition (\texttt{demo/cassandra/schema.cql}):}
\begin{lstlisting}[language=SQL]
CREATE KEYSPACE IF NOT EXISTS shop
  WITH replication = {
    'class': 'SimpleStrategy',
    'replication_factor': 1
  };

CREATE TABLE IF NOT EXISTS shop.products_by_category (
  category     text,
  price        decimal,
  product_id   int,
  name         text,
  brand        text,
  description  text,
  rating       float,
  stock        int,
  review_count int,
  PRIMARY KEY ((category), price, product_id)
) WITH CLUSTERING ORDER BY (price ASC, product_id ASC);
\end{lstlisting}
\end{column}
\begin{column}{0.45\textwidth}
\begin{block}{\scriptsize Anatomy of the Compound Primary Key}
\scriptsize
\texttt{PRIMARY KEY ((category), price, product\_id)}
\begin{itemize}
    \item \textbf{Partition Key \texttt{(category)}:} Determines \emph{which node/partition} stores the data via $\text{Murmur3}(\texttt{category})$. All laptops live together in one physical partition.
    \item \textbf{Clustering Columns \texttt{(price, product\_id)}:} Determines the \emph{on-disk sort order} within that partition.
    \item \texttt{product\_id} acts as tie-breaker so two products with the same price don't overwrite each other.
\end{itemize}
\end{block}
\scriptsize
\textbf{Result:} Category browse + price range filter is a \textbf{single sequential disk read} with zero secondary index overhead!
\end{column}
\end{columns}
\end{frame}

  ON shop.products_by_category (rating)
  USING 'StorageAttachedIndex';

\begin{block}{\scriptsize How SAI Works Under the Hood \cite{sai}}
\scriptsize
\begin{itemize}
    \item \textbf{SSTable-Attached Lifecycle:} Index files are written alongside SSTable flushes and merged during compaction.
    \item \textbf{Strings (\texttt{brand}):} Indexed via memory-efficient \textbf{Byte-Ordered Tries}.
    \item \textbf{Numerics (\texttt{rating}):} Indexed via balanced \textbf{BKD Trees} for range lookups.
\end{itemize}
\end{block}
\end{column}

\midrule

Without a partition key (\texttt{WHERE brand='Sony'}), SAI still performs a \textbf{scatter-gather} across ring nodes, though local SSTable lookup is much faster than full scan.
\end{alertblock}

% SLIDE 16: Cassandra Limitations for Search

\begin{frame}[fragile]
\frametitle{Cassandra Limitations for Full-Text Search}
\begin{columns}[T]
\begin{column}{0.52\textwidth}
\textbf{\scriptsize 1. Refusal on Unindexed Columns (Demo 5)}
\begin{lstlisting}[language=SQL]
SELECT name FROM shop.products_by_category
WHERE stock > 100 LIMIT 5;
-- InvalidRequest: Cannot execute this query as it
-- might involve data filtering... use ALLOW FILTERING
\end{lstlisting}

\textbf{\scriptsize 2. The \texttt{ALLOW FILTERING} Anti-Pattern (Demo 6)}
\begin{lstlisting}[language=SQL]
SELECT name, stock FROM shop.products_by_category
WHERE stock > 100 LIMIT 5 ALLOW FILTERING;
\end{lstlisting}
\scriptsize
\begin{itemize}
    \item Works on 5,000 demo rows, but in production forces coordinator to scan \textbf{every SSTable across every node}.
\end{itemize}

\textbf{\scriptsize 3. No \texttt{LIKE '\%...\%'} Substring Search (Demo 7)}
\begin{lstlisting}[language=SQL]
SELECT name FROM shop.products_by_category
WHERE description LIKE '%laptop%' LIMIT 3;
-- InvalidRequest: LIKE restriction is only supported
-- on properly indexed columns.
\end{lstlisting}
\end{column}
\begin{column}{0.45\textwidth}
\begin{alertblock}{\scriptsize What Cassandra Cannot Do Natively}
\scriptsize
\begin{itemize}
    \item \textbf{No Relevance Ranking (BM25):} Rows are returned strictly in token or clustering order --- no concept of ``best match''.
    \item \textbf{No Fuzzy / Typo Tolerance:} Cannot match \texttt{"lapptop"} to \texttt{"laptop"}.
    \item \textbf{No Stemming or Synonyms:} \texttt{"cameras"} will not match \texttt{"camera"} without custom analyzer pipelines.
    \item \textbf{No Faceted Aggregations:} Computing \texttt{GROUP BY category} with \texttt{AVG(price)} across partitions is an anti-pattern.
\end{itemize}
\end{alertblock}
\vspace{0.05cm}
\scriptsize
\textbf{Key Insight:} Cassandra's refusal is a \emph{feature}, protecting cluster SLA from unbounded latency.
\end{column}
\end{columns}
\end{frame}

\textbf{\scriptsize 3. Forcing \texttt{ALLOW FILTERING} (Demo 6)}
\begin{lstlisting}[language=SQL]
CQL> SELECT name, stock
     FROM shop.products_by_category
     WHERE stock > 100 LIMIT 5
     ALLOW FILTERING;

 name                     | stock
--------------------------+-------
 Breville BrewPerfect 454 |   488
 DeLonghi BrewPerfect 161 |   175
  Philips BrewPerfect 968 |   332
    Nespresso MokaPro 354 |   456
     Breville MokaPro 381 |   253
\end{lstlisting}
\begin{alertblock}{\scriptsize Why This Matters in Production}
\scriptsize
\begin{itemize}
    \item \texttt{ALLOW FILTERING} scans all partitions until \texttt{LIMIT 5} is satisfied (or scans 100\% of the cluster if $<5$ rows match).
    \item \textbf{Takeaway:} Never use Cassandra for ad-hoc text search; pair it with Elasticsearch!
\end{itemize}
\end{alertblock}

\begin{frame}
\frametitle{Conclusion \& Key Takeaways}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{block}{\scriptsize 1. Elasticsearch = Discovery \& Relevance}
\scriptsize
\begin{itemize}
    \item Answers \textbf{unstructured, user-typed queries} via Lucene inverted indexes, custom analyzers (stemming, stopwords, edge n-grams), and FSTs.
    \item Provides \textbf{Okapi BM25 ranking}, typo-tolerant fuzzy search, highlighting, and real-time faceted aggregations via columnar \texttt{doc\_values}.
\end{itemize}
\end{block}
\begin{block}{\scriptsize 2. Cassandra = Predictable Scale}
\scriptsize
\begin{itemize}
    \item Answers \textbf{known, structured access patterns} in $O(1)$ network hops via consistent hashing and sorted LSM-Tree partitions.
    \item Cassandra 5.0 \textbf{SAI} enables efficient secondary attribute filtering, but purposefully omits full-text relevance ranking.
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{alertblock}{\scriptsize 3. Architectural Synthesis}
\scriptsize
\begin{itemize}
    \item \textbf{Right Tool for the Job:} Neither database replaces the other; their internal data structures (\emph{Inverted Index vs. Partitioned LSM-Tree}) are optimized for fundamentally different workloads.
    \item \textbf{Production Standard:} Combine \textbf{Cassandra as the durable source of truth} with \textbf{Elasticsearch as the search index}, synchronized asynchronously via Change Data Capture (CDC) and Kafka.
\end{itemize}
\end{alertblock}
\vspace{0.1cm}
\centering
\scriptsize
\textbf{Reproducible Artifact:} Full Docker stack, 5,000-product generator, ES mappings, CQL schema, and 17 live demo queries included in repository.
\end{column}
\end{columns}
